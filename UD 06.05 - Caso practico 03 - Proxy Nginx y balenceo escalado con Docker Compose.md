# UD 06.05 - Caso practico 03 - Proxy Nginx y balenceo escalado con Docker Compose

# 0-Crear el directorio del proyecto

mkdir nginx-balanceo

cd nginx-balanceo

# 1-Crear estructura de directorios

mkdir apache

mkdir nginxproxy

# 3-Crear el Dockerfile de Apache

nano apache/Dockerfile

Contenido:

FROM php:7.2-apache

COPY index.php /var/www/html/

# 4-Crear el fichero index.php

nano apache/index.php

Contenido:

<html>
  
<body>
  
<h1>
  
Servido por: Servidor con IP <?php echo $_SERVER['SERVER_ADDR']; ?>

y hostname <?php echo gethostname(); ?>

</h1>

</body>

</html>

# 5-Crear el fichero de configuración de Nginx

nano nginxproxy/nginx.conf

Contenido:

user nginx;

events {

    worker_connections 1000;
    
}

http {

    server {
    
        listen 4000;
        
        location / {
        
            proxy_pass http://apache:80;
            
        }
        
    }
    
}

# 6-Crear el fichero docker-compose.yml

nano docker-compose.yml

Contenido:

version: "3.9"

services:

  apache:
  
    build: ./apache
    
    restart: always
    
    ports:
    
      - "80"

  nginxproxy:
  
    image: nginx:latest
    
    volumes:
    
      - ./nginxproxy/nginx.conf:/etc/nginx/nginx.conf:ro
      
    depends_on:
    
      - apache
      
    ports:
    
      - "4000:4000"

# 7-Poner en marcha el sistema

docker compose up -d

# 8-Comprobar contenedores

docker compose ps

# 9-Acceder al proxy

Abrir navegador:

http://localhost:4000

# 10-Escalar servidores Apache

docker compose up -d --scale apache=4

# 11-Comprobar de nuevo

docker compose ps

# 12-Probar el balanceo

En el navegador:

http://localhost:4000

# 13-Parar el sistema

docker compose down
